Quick start 
import { DesmakeClient } from '@desmake/sdk';

const client = new DesmakeClient({
  apiKey: process.env.DESMAKE_API_KEY
});

// Search the catalogue
const results = await client.catalog.search({
  q: 'minimal poster',
  limit: 12,
  adapters: ['poster', 'tshirt']
});

// Generate a new design via AI
const design = await client.design.generate({
  prompt: 'brutalist geometric poster in ember palette',
  style: 'abstract'
});

// Publish it through D2M engine
const listing = await client.design.publish({
  designId: design.id,
  adapters: ['poster', 'tshirt', 'sticker']
});
from desmake import DesmakeClient

client = DesmakeClient(api_key=os.environ["DESMAKE_API_KEY"])

# Search the catalogue
results = client.catalog.search(
    q="minimal poster",
    limit=12,
    adapters=["poster", "tshirt"]
)

# Generate a new design via AI
design = client.design.generate(
    prompt="brutalist geometric poster in ember palette",
    style="abstract"
)

# Publish it through D2M engine
listing = client.design.publish(
    design_id=design.id,
    adapters=["poster", "tshirt", "sticker"]
)
# Search the catalogue
curl -X GET "https://api.desmake.com/v1/catalog/search" \
  -H "Authorization: Bearer $DESMAKE_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"q":"minimal poster","limit":12}'

# Generate a design
curl -X POST "https://api.desmake.com/v1/design/generate" \
  -H "Authorization: Bearer $DESMAKE_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"prompt":"brutalist geometric poster"}'

# Publish to marketplace
curl -X POST "https://api.desmake.com/v1/design/publish" \
  -H "Authorization: Bearer $DESMAKE_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"design_id":"dsg_xxx","adapters":["poster","tshirt"]}'
